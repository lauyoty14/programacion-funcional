data EA = Const Int | BOp BinOp EA EA      
data BinOp = Sum | Mul

a)
foldEA :: (Int -> b) -> (BinOp -> b -> b -> b) -> EA -> b 
foldEA fc fo (Const i) = fc i 
foldEA fc fo (BinOp op e1 e2) = fo op (foldEA fc fo e1) (foldEA fc fo e2)

b) i)
noTieneNegativosExplicitosEA  ::  EA  ->  Bool
noTieneNegativosExplicitosEA = foldEA (>=0) (\op b1 b2 -> b1 && b2)

ii)
simplificarEA' :: EA -> EA
simplificarEA' = foldEA 
    (Const) 
    (\op b1 b2 -> case op of 
                    Sum -> simplificarSum b1 b2 
                    Mul -> simplificarMul b1 b2)

simplificarSum :: EA -> EA -> EA 
simplificarSum (Const 0) ea2 = ea2
simplificarSum ea1 (Const 0) = ea1
simplificarSum ea1 ea2 = BOp Sum ea1 ea2

simplificarMul :: EA -> EA -> EA 
simplificarMul (Const 0) ea2 = Const 0
simplificarMul ea1 (Const 0) = Const 0
simplificarMul (Const 1) ea2 = ea2
simplificarMul ea1 (Const 1) = ea1
simplificarMul ea1 ea2 = BOp Mul ea1 ea2

iii)
evalEA'  :: EA -> Int
evalEA' = foldEA 
    (id)
    (\op n m -> case op of 
                    Sum -> n + m 
                    Mul -> n * m)

iv)

                (\op r1 r2 -> case op of
                                Sum -> "(" ++ r1 ++ "+" ++ r2 ++ ")"
                                Mul -> "(" ++ r1 ++ "*" ++ r2 ++ ")")
