cantidadDePuntosVacios :: Dungeon a -> Int
cantidadDePuntosVacios (Habitacion _) = 0
cantidadDePuntosVacios (Pasaje a d) = 
    case a of
        Just _ -> cantidadDePuntosVacios d
        Nothing -> 1 + cantidadDePuntosVacios d                                     
cantidadDePuntosVacios (Bifurcacion a d d') =
    case a of
        Just _ -> cantidadDePuntosVacios d + cantidadDePuntosVacios d'
        Nothing -> 1 + cantidadDePuntosVacios d + cantidadDePuntosVacios d'

cantidadDePuntosCon :: a -> Dungeon a -> Int
cantidadDePuntosCon a (Habitacion n) = if a == n 
                                        then 1
                                        else 0
cantidadDePuntosCon a (Pasaje n d) = 
    case n of
        Nothing -> cantidadDePuntosCon a d
        Just m -> if a == m
                    then 1 + cantidadDePuntosCon a d
                  else cantidadDePuntosCon a d
cantidadDePuntosCon a (Bifurcacion n d d') = 
    case n of 
        Nothing -> cantidadDePuntosCon a d + cantidadDePuntosCon a d'
        Just m -> if a == m
                    then 1 + cantidadDePuntosCon a d + cantidadDePuntosCon a d' 
                  else cantidadDePuntosCon a d + cantidadDePuntosCon a d'

esLineal :: Dungeon a -> Bool
esLineal (Habitacion _) = True
esLineal (Pasaje _ d) = esLineal d
esLineal (Bifurcacion _ _ _) = False
f)
llenoDe :: a -> Dungeon a -> Bool
llenoDe x (Habitacion a) = x == a 
llenoDe x (Pasaje a d) = 
    case x of
        Nothing -> False
        Just y -> y == a && llenoDe x d
llenoDe x (Bifurcacion a d d') = 
    case x of
        Nothing -> False 
        Just y -> y == a && llenoDe x d && llenoDe x d'
