type Potencia = Int
type Ataque = Int

data Red = Terminal Potencia
         | Servidor Potencia Red
         | Firewall (Ataque -> Bool) Red
         | Switch (Ataque -> Dir) Red Red

data Dir = Izq | Der | Forward
type Ruta = [Dir]

applyFst :: (a -> a) -> (a, b) -> (a, b)
applyFst f (n, y) = (f n, y)

combinarFirewalls :: Red -> Red
combinarFirewalls (Terminal x) =
	Terminal x
combinarFirewalls (Servidor x r) =
	Servidor x (combinarFirewalls r)
combinarFirewalls (Firewall fb r) =
	juntarFirewalls fb (combinarFirewalls r)
combinarFirewalls (Switch fd ri rd) =
	Switch fd (combinarFirewalls ri)
	          (combinarFirewalls rd)

juntarFirewalls :: (Ataque -> Bool) -> Red -> Red
juntarFirewalls fb1 (Firewall fb2 r) =
	Firewall (\a -> fb1 a && fb2 a) r
juntarFirewalls fb1 r =	Firewall fb1 r

----------------------------------------------------------------

intensidadRestante :: Ataque -> Red -> Maybe Ataque
intensidadRestante n (Terminal x) =
	Just (n - x)
intensidadRestante n (Servidor x r) =
	intensidadRestante (n + x) r
intensidadRestante n (Firewall fb r) =
	if fb n
	   then intensidadRestante n r
	   else Nothing
intensidadRestante n (Switch fd ri rd) =
	case fd n of
		Der -> case intensidadRestante n rd of
			        Nothing -> Nothing
			        Just n' -> intensidadRestante n' ri
		_   -> case intensidadRestante n ri of
			        Nothing -> Nothing
			        Just n' -> intensidadRestante n' rd

pasosHastaBloqueo :: Ataque -> Red -> Maybe Int
pasosHastaBloqueo n r = 
	case pasosRec n r of
		(n, Nothing) -> Just n
		(_, _)       -> Nothing

pasosRec :: Ataque -> Red -> (Int, Maybe Ataque)
pasosRec n (Terminal x) =
	(1, Just (n - x))
pasosRec n (Servidor x r) =
	applyFst (+1) (pasosRec (n + x) r)
pasosRec n (Firewall fb r) =
	if fb n
	   then applyFst (+1) (pasosRec n r)
	   else (0, Nothing)

pasosRec n (Switch fd ri rd) =
	case fd n of
		Der -> case pasosRec n rd of
			        (pasosD, Nothing) -> (pasosD + 1, Nothing)
			        (pasosD, Just n') -> let (pasosI, m) = pasosRec n' ri
			                                 in (pasosI + pasosD + 1, m)

		_   -> case pasosRec n ri of
			        (pasosI, Nothing) -> (pasosI + 1, Nothing)
			        (pasosI, Just n') -> let (pasosD, m) = pasosRec n' ri
			                                 in (pasosI + pasosD + 1, m)

rutasAServidores :: Ataque -> Red -> [Ruta]
rutasAServidores n r = fst (rutasRec n r)

rutasRec :: Ataque -> Red -> ([Ruta], Maybe Ataque)
rutasRec n (Terminal x) =
	([], Just (n - x))

rutasRec n (Servidor x r) =
	let (rutas, m) = rutasRec (n + x) r
	    in ([] : map (Forward :) rutas, m)
rutasRec n (Firewall fb r) =
	if fb n
	   then let (rutas, m) = rutasRec n r
	            in (map (Forward :) rutas, m)
	   else ([], Nothing)
rutasRec n (Switch fd ri rd) =
	case fd n of
		Der -> case rutasRec n rd of
			        (rutasD, Nothing) -> (map (Der :) rutasD, Nothing)
			        (rutasD, Just n') -> let (rutasI, m) = rutasRec n' ri
			                                 in (map (Der :) rutasD ++ map (Izq :) rutasI, m)
		_   -> case rutasRec n ri of
			        (rutasI, Nothing) -> (map (Izq :) rutasI, Nothing)
			        (rutasI, Just n') -> let (rutasD, m) = rutasRec n' ri
			                                 in (map (Izq :) rutasI ++ map (Der :) rutasD, m)

foldRed :: (Potencia -> b) 
	-> (Potencia -> b -> b) 
	-> ((Ataque -> Bool) -> b -> b)
	-> ((Ataque -> Dir) -> b -> b -> b)
	-> Red
	-> b
foldRed ft fsrv ffr fstw (Terminal x) =
	ft x
foldRed ft fsrv ffr fstw (Servidor x r) =
	fsrv x (foldRed ft fsrv ffr fstw r)
foldRed ft fsrv ffr fstw (Firewall fb r) =
	ffr fb (foldRed ft fsrv ffr fstw r)
foldRed ft fsrv ffr fstw (Switch fd ri rd) =
	fstw fd (foldRed ft fsrv ffr fstw ri)
	        (foldRed ft fsrv ffr fstw rd)

recRed :: (Potencia -> b) 
	-> (Potencia -> Red -> b -> b) 
	-> ((Ataque -> Bool) -> Red -> b -> b)
	-> ((Ataque -> Dir) -> Red -> Red -> b -> b -> b)
	-> Red
	-> b
recRed ft fsrv ffr fstw (Terminal x) =
	ft x
recRed ft fsrv ffr fstw (Servidor x r) =
	fsrv x r (recRed ft fsrv ffr fstw r)
recRed ft fsrv ffr fstw (Firewall fb r) =
	ffr fb r (recRed ft fsrv ffr fstw r)
recRed ft fsrv ffr fstw (Switch fd ri rd) =
	fstw fd ri rd (recRed ft fsrv ffr fstw ri) (recRed ft fsrv ffr fstw rd)
