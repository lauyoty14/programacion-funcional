data Accion a = Paso a | SaltoArriba a | SaltoAdelante a | Girar a   
type Tiempo = Int     -- el instante en el que sucede un movimiento   
type Duracion = Int   -- cantidad de tiempos que dura el movimiento   
data Animacion a =   
     Espera Duracion                       -- durante la duración dada no hay acciones   
   | Mov Duracion (Accion a)               -- un cierto movimiento con una duración dada   
   | Sec (Animacion a) (Animacion a)       -- secuencia (la 2da empieza al terminar la 1era)   
   | Par (Animacion a) (Animacion a)       -- paralelo (arrancan juntas y dura lo que la más larga)

type Frame a = [Accion a]     -- acciones simultáneas en un tiempo específico   
type Simulador a = Tiempo -> Frame a -- función que da las acciones que ocurren en un tiempo dado

combinarSinDuplicados :: [Int] -> [Int] -> [Int]
combinarSinDuplicados [] ys = ys
combinarSinDuplicados xs [] = xs
combinarSinDuplicados (x:xs) (y:ys) = 
  if x == y
    then x : (combinarSinDuplicados xs ys)
    else if x < y 
            then x : (combinarSinDuplicados xs (y:ys))
            else y : (combinarSinDuplicados (x:xs) ys)

simular :: Animacion a -> [Frame a]
simular (Espera d) = replicate d []
simular (Mov d ac) = replicate d [ac]
simular (Sec ani1 ani2) = (simular ani1) ++ (simular ani2)
simular (Par ani1 ani2) = simularPar (simular ani1) (simular ani2)

simularPar :: [Frame a] -> [Frame a] -> [Frame a]
simularPar [] fs = fs
simularPar fs [] = fs
simularPar (f1:fs1) (f2:fs2) = (f1 ++ f2) : (simularPar fs1 fs2)

tiemposDeEspera :: Animacion a -> [Tiempo]
tiemposDeEspera (Espera d) = contarHasta d 
tiemposDeEspera (Mov d ac) = []
tiemposDeEspera (Sec ani1 ani2) = tiemposDeEspera ani1 ++ (map (+ (duracion ani1)) tiemposDeEspera ani2)
tiemposDeEspera (Par ani1 ani2) = combinarSinDuplicados (tiemposDeEspera ani1) (tiemposDeEspera ani2)

contarHasta :: Int -> [Int]
contarHasta 0 = []
contarHasta n = contarHasta (n - 1) ++ [n]

foldAni :: (Duracion -> b) -> (Duracion -> Accion a -> b) -> (b -> b -> b) -> (b -> b -> b) -> Animacion a -> b 
foldAni fe _ _ _ (Espera d) = fe d 
foldAni _ fm _ _ (Mov d ac) = fm d ac
foldAni fe fm fs fp (Sec ani1 ani2) = fs (foldAni fe fm fs fp ani1) (foldAni fe fm fs fp ani2)
foldAni fe fm fs fp (Par ani1 ani2) = fp (foldAni fe fm fs fp ani1) (foldAni fe fm fs fp ani2)

(b -> Animacion a -> b -> Animacion a -> b) -> Animacion a -> b
recAni fe _ _ _ (Espera d) = fe d 
recAni _ fm _ _ (Mov d ac) = fm d ac
recAni fe fm fs fp (Sec ani1 ani2) = fs (recAni fe fm fs fp ani1) ani1 (recAni fe fm fs fp ani2) ani2
recAni fe fm fs fp (Par ani1 ani2) = fp (recAni fe fm fs fp ani1) ani1 (recAni fe fm fs fp ani2) ani2
