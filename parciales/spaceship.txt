wreck :: Hazard -> Spaceship -> Spaceship
wreck (d , n , _) spaceship = impactInDirectionLevel d n spaceship

impactInDirectionLevel :: Direction -> Int -> Spaceship -> Spaceship
impactInDirectionLevel _ _ Plug = Plug
impactInDirectionLevel Larboard n (Module c s1 s2) =
                            if n <= 1
                                then Plug
                                else Module c (impactInDirectionLevel Larboard (n - 1) s1) s2
impactInDirectionLevel Starboard n (Module c s1 s2) =
                            if n <= 1
                                then Plug
                                else Module c s1 (impactInDirectionLevel Starboard (n - 1) s2)

rectSS f z (Module c s1 s2) = f c s1 (foldSS f z s1) s2 (foldSS f z s2)

manoeuvre :: Spaceship -> [Hazard] -> Spaceship
manoeuvre ship hs = foldr (\h manoShip -> let (dir, level, size) = h
                                        escudada = shielded manoShip in
                                case size of 
                                    Small -> if escudada
                                                then manoShip
                                            else wreck h manoShip
                                    Big -> if armed manoShip
                                            then if escudada 
                                                    then manoShip
                                                    else wreck (dir, level, Small) manoShip
                                            else wreck h manoShip
                                    Torpedo -> wreck h manoShip) ship hs
