occursMESWith :: a -> (a->a) -> MSExp a -> Int
occursMESWith a f (EmptyMS)         = 0
occursMESWith a f (AddMS b ms)      = if a == (f b) 
                                        then 1 + occursMESWith a f ms
                                        else occursMESWith a f ms
occursMESWith a f (UnionMS ms1 ms2) = occursMESWith a f ms1 + occursMESWith a f ms2
occursMESWith a f (RemoveMS b ms)   = if a == (f b)
                                        then max 0 (occursMESWith a f ms - 1)
                                        else occursMESWith a f ms
occursMESWith a f (MapMS g ms)      = occursMESWith a (f . g) ms

filterMSEif f (EmptyMS)         = EmptyMS
filterMSEif f (AddMS x ms)      = if f x
                                    then AddMS x (filterMSEif f ms)
                                    else filterMSEif f ms
filterMSEif f (UnionMS ms1 ms2) = UnionMS (filterMSEif f ms1) (filterMSEif f ms2)
filterMSEif f (RemoveMS x ms)   = if f x
                                    then RemoveMS x (filterMSEif f ms)
                                    else filterMSEif f ms
filterMSEif f (MapMS g ms)      = MapMS g (filterMSEif (f . g) ms)

evalMSE :: Eq a => MSExp a -> [a]
evalMSE EmptyMS         = []
evalMSE (AddMS x e)     = x : evalMSE e
evalMSE (RemoveMS x e)  = listWithout x (evalMSE e)
evalMSE (UnionMS e1 e2) = evalMSE e1 ++ evalMSE e2
evalMSE (MapMS f e)     = map f (evalMSE e)

listWithout :: Eq a => a -> [a] -> [a]
listWithout x [] = error "no se puede sacar"
listWithout x (y:ys) = if x == y then ys else y : listWithout x ys

simpMSE :: MSExp a -> MSExp a
simpMSE EmptyMSE        = EmptyMSE
simpMSE (AddMS x e)     = AddMS x (simpMSE e)
simpMSE (RemoveMS x e)  = simpRemoveMS x (simpMSE e)
simpMSE (UnionMS e1 e2) = simpUnionMS (simpMSE e1) (simpMSE e2)
simpMSE (MapMS f e)     = simpMapMS f (simpMSE e)

simpRemoveMS Eq a => a -> MSExp a -> MSExp a
simpRemoveMS x (AddMS y e) = if x==y
                                then e
                                else RemoveMS x (AddMS y e)
simpRemoveMS x e           = RemoveMS x e

simpUnionMS :: MSExp a -> MSExp a -> MSExp a
simpUnionMS EmptyMS e2 = e2
simpUnionMS e1 EmptyMS = e1
simpUnionMS e1      e2 = UnionMS e1 e2

simpMapMS :: (a -> a) -> MSExp a -> MSExp a
simpMapMS f EmptyMS = EmptyMS
simpMapMS f e       = MapMS f e

foldMSE :: b -> (a -> b -> b) -> (b -> b -> b) -> (a -> b -> b) -> ((a -> a) -> b -> b) -> MSExp a -> b
foldMSE em _ _ _ _ EmptyMS                    = em 
foldMSE em add union rem ma (AddMS a ms)      = add a (foldMSE em add union rem ma ms)
foldMSE em add union rem ma (UnionMS ms1 ms2) = union (foldMSE em add union rem ma ms1) 
                                                      (foldMSE em add union rem ma ms2)
foldMSE em add union rem ma (RemoveMS a ms)   = rem a (foldMSE em add union rem ma ms)
foldMSE em add union rem ma (MapMS f ms)      = ma f (foldMSE em add union rem ma ms)

                      (\b or f -> if a == (f b) then 1 + or f else or f)
                      (\or1 or2 f -> or1 f + or2 f)
                      (\b or f -> if a == (f b) then max 0 (or f - 1) else or f) 
                      (\g orG f -> orG (f . g)) msExp id

filterMSE f msExp = foldMSE (const EmptyMS)
                    (\a ms p -> if p a then AddMS a (ms p) else ms p)
                    (\ms1 ms2 p -> UnionMS (ms1 p) (ms2 p))
                    (\a ms p -> if p a then RemoveMS a (ms p) else ms p)
                    (\g ms p -> MapMS g (ms (p . g))) msExp f
