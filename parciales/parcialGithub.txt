type Distancia = Int
type Peso      = Int
type Altura    = Int

data Sector
    = Base Distancia
    | Trayecto Distancia Sector
    | Escudo [Altura] Sector
    | Conexión Peso Sector Sector

ejercicio 1)
cantidadDeConexiones :: Sector -> Int
cantidadDeConexiones (Base d) = 0
cantidadDeConexiones (Trayecto d sec) = cantidadDeConexiones sec
cantidadDeConexiones (Escudo als sec) = cantiadDeContexiones sec
cantidadDeConexiones (Conexion p secI secD) = 1 + cantiadDeContexiones secI + cantiadDeContexiones secD

ejercicio 2)
foldSector :: (Distancia -> b) -> (Distancia -> b -> b) -> ([Altura] -> b -> b) -> (Peso -> b -> b -> b) -> Sector -> b
foldSector fb _ _ _ (Base d) = fb d
foldSector fb ft fe fc (Trayecto d sec) = ft d (foldSector fb ft fe fc sec)
foldSector fb ft fe fc (Escudo alts sec) = fe alts (foldSector fb ft fe fc sec) 
foldSector fb ft fe fc (Conexion p secI secD) = fc p (foldSector fb ft fe fc secI) (foldSector fb ft fe fc secD)

ejercicio 3) 
recSector :: (Distancia -> b) -> 
        (Distancia -> Sector -> b -> b) -> 
        ([Altura] -> Sector -> b -> b) -> 
        (Peso -> Sector -> b -> Sector -> b -> b) -> Sector -> b
recSector fb _ _ _ (Base d) = fb d
recSector fb ft fe fc (Tratecto d sec) = ft d sec (recSector fb ft fe fc sec)
recSector fb ft fe fc (Escudo alts sec) = fe alts sec (recSector fb ft fe fc sec)
recSector fb ft fe fc (Conexion p secI secD) = fc p secI (recSector fb ft fe fc secI) secD (recSector fb ft fe fc secD)

ejercicio 4)
alturasCríticas :: Sector -> [Altura]
alturasCriticas = foldSector 
                    (const []) 
                    (\_ altsR -> altsR)
                    (\alts altsR -> if length alts > 50 
                                        then alts ++ altsR
                                        else altsR)
                    (\p altsRI altsRD -> altsRI ++ altsRD)

ejercicio 5)
optimizarSector :: Sector -> Sector
optimizarSector = recSector 
                    (Base)
                    (\d sec secR -> Trayecto d secR)
                    (\alts sec secR -> if null alts 
                                        then secR
                                        else Escudo alts secR)
                    (\p secI secIR secD secDR -> if p == 0
                                                    then secIR
                                                    else Conexion p secIR secDR)
