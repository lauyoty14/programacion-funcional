data Dir = Left | Right | Straight
data Mapa a = Cofre [a] | Nada (Mapa a) | Bifurcacion [a] (Mapa a) (Mapa a)

hasObjectAt :: (a -> Bool) -> Mapa a -> [Dir] -> Bool
hasObjectAt f (Cofre as) dir = 
    case dir of 
        [] -> any f as
        (d:ds) -> error "camino invalido"
hasObjectAt f (Nada m) dir =
    case dir of 
        [] -> error "camino invalido"
        (d:ds) -> case d of 
                    Straight -> hasObjectAt f m ds
                    _ -> error "camino invalido"
hasObjectAt f (Bifurcacion as mapI mapD) dir = 
    case dir of 
        [] -> any f as
        (d:ds) -> case d of 
                    Left -> hasObjectAt f mapI ds
                    Right -> hasObjectAt f mapD ds 
                    _ -> error "camino invalido"

allPaths :: Mapa a -> [[Dir]]
allPaths (Cofre _) = [[]]
allPaths (Nada m)  = map (Straight:) (allPaths m)
allPaths (Bifurcacion _ mapI mapD) = (map (Left:) (allPaths mapI)) ++ (map (Right:) (allPaths mapD))

foldM :: ([a] -> b) -> (b -> b) -> ([a] -> b -> b -> b) -> Mapa a -> b
foldM fc _ _ (Cofre as) = fc as
foldM fc fn fb (Nada m) = fn (foldM fc fn fb m)
foldM fc fn fb (Bifurcacion as mapI mapD) = fb as (foldM fc fn fb mapI) (foldM fc fn fb mapD)

recM :: ([a] -> b) -> (Mapa a -> b -> b) -> ([a] -> Mapa a -> b -> Mapa a -> b -> b) -> Mapa a -> b
recM fc _ _ (Cofre as) = fc as
recM fc fn fb (Nada m) = fn m (recM fc fn fb m)
recM fc fn fb (Bifurcacion as mapI mapD) = fb as mapI (recM fc fn fb mapI) mapD (recM fc fn fb mapD)
