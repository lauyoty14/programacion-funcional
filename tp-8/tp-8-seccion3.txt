data ExpA = Cte Int
| Suma ExpA ExpA
| Prod ExpA ExpA

a)
evalExpA :: ExpA -> Int
evalExpA (Cte i)      = i
evalExpA (Suma e1 e2) = evalExpA e1 + evalExpA e2
evalExpA (Prod e1 e2) = evalExpA e1 * evalExpA e2
b)
simplificarExpA :: ExpA -> ExpA
simplificarExpA (Cte i)      = (Cte i)
simplificarExpA (Suma e1 e2) = sumaSinNeutro (simplificarExpA e1) (simplificarExpA e2)   
simplificarExpA (Prod e1 e2) = prodSinNeutroNiCero (simplificarExpA e1) (simplificarExpA e2)

sumaSinNeutro :: ExpA -> ExpA -> ExpA
sumaSinNeutro (Cte 0) e2 = e2
sumaSinNeutro e1 (Cte 0) = e1
sumaSinNeutro e1 e2 = Suma e1 e2

prodSinNeutroNiCero :: ExpA -> ExpA -> ExpA
prodSinNeutroNiCero (Cte 1) e2 = e2
prodSinNeutroNiCero e1 (Cte 1) = e1
prodSinNeutroNiCero (Cte 0) _ = Cte 0
prodSinNeutroNiCero _ (Cte 0) = Cte 0
prodSinNeutroNiCero e1 e2 = Prod e1 e2
c)
cantidadDeSumaCero :: ExpA -> Int
cantidadDeSumaCero (Cte i)      = 0
cantidadDeSumaCero (Suma e1 e2) = unoSiEsSumaCero e1 e2 + cantidadDeSumaCero e1 + cantidadDeSumaCero e2 
cantidadDeSumaCero (Prod e1 e2) = cantidadDeSumaCero e1 + cantidadDeSumaCero e2

unoSiEsSumaCero :: ExpA -> ExpA -> Int
unoSiEsSumaCero (Cte 0) _ = 1  
unoSiEsSumaCero _ (Cte 0) = 1
unoSiEsSumaCero e1 e2 = 0
